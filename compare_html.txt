            if (e && e.target && e.target !== document.getElementById('compareModal') && !e.target.closest('.compareModalClose')) {
                return;
            }

        // FILTER HANDLERS

        function handleCapitalFilter(val) {
            currentFilterCapital = val;
            renderPlans();
        }

        function handlePriorityFilter(val) {
            currentFilterPriority = val;
            document.querySelectorAll('.toolPill').forEach(btn => {
                btn.classList.toggle('active', btn.dataset.pill === val);
            });
            renderPlans();
        }

        function handlePillFilter(val) {
            currentFilterPriority = val;
            document.querySelectorAll('.toolPill').forEach(btn => {
                btn.classList.toggle('active', btn.dataset.pill === val);
            });
            const radio = document.querySelector(`input[name="priorityRadio"][value="${val}"]`);
            if (radio) radio.checked = true;
            renderPlans();
        }

        function openLeadModalForPlan(planTitle) {
            const modal = document.getElementById('leadModal');
            if (!modal) return;
            const desc = modal.querySelector('#leadContent p');
            if (desc) {
                desc.innerHTML = `Déjanos tus datos y un Agente especialista de MHM te contactará sobre <strong>${planTitle}</strong> con una propuesta a tu medida.`;
            }
            modal.classList.add('active');
        }

        function closeLeadModal() {
            const modal = document.getElementById('leadModal');
            if (modal) modal.classList.remove('active');
        }

        // Contratar Plan Checkout Flow
        function contratarPlan(planId) {
            const plans = getMapfrePlans();
            const planDef = plans.find(p => p.id === planId) || plans[0];
            const totalMembers = 1 + (profile.partner ? 1 : 0) + (profile.children || 0);

            sessionStorage.setItem('mhm_salud_plan', JSON.stringify({
                id: planId,
                name: planDef.title,
                tier: planDef.tier,
                capitalUF: planDef.capitalUF,
                price: planDef.rawPrice,
                priceFormatted: planDef.priceMonth,
                priceYear: planDef.priceYear,
                planType: planDef.planType || (planId.includes('familiar') ? 'familiar' : 'individual')
            }));

            sessionStorage.setItem('mhm_salud_profile', JSON.stringify({
                age: profile.age,
                gender: profile.gender,
                health: profile.health,
                partner: profile.partner,
                children: profile.children,
                totalMembers: totalMembers
            }));

            window.location.href = 'cotizacion-salud-resumen.html';
        }
